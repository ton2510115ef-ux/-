const buttons = document.querySelectorAll('.company-btn');
const detailBox = document.getElementById('company-detail');
const detailName = document.getElementById('detail-name');
const detailAddress = document.getElementById('detail-address');
const detailInfo = document.getElementById('detail-info');

buttons.forEach(button => {
    button.addEventListener('click', () => {
       
        detailName.textContent = button.getAttribute('data-name');
        detailAddress.textContent = button.getAttribute('data-address');
        detailInfo.textContent = button.getAttribute('data-info');
        
        
        detailBox.classList.remove('hidden');
    });
});


const downloadBtn = document.getElementById('download-btn');
downloadBtn.addEventListener('click', () => {
    
    let csvContent = "\uFEFF"; // 文字化け防止のBOM
    csvContent += "会社名,住所,その他情報\n"; // ヘッダー

    buttons.forEach(button => {
        const name = button.getAttribute('data-name');
        const address = button.getAttribute('data-address');
        const info = button.getAttribute('data-info');
        csvContent += `"${name}","${address}","${info}"\n`;
    });

   
    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'companies.csv';
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
});
